import React, { useState } from 'react';
import { 
  Scale, 
  CheckCircle2, 
  XCircle, 
  Table, 
  Grid2X2, 
  Sparkles, 
  ArrowRight,
  Loader2 
} from 'lucide-react';

export default function TheTiebreaker() {
  const [decisionInput, setDecisionInput] = useState('');
  const [loading, setLoading] = useState(false);
  const [activeTab, setActiveTab] = useState<'pros_cons' | 'matrix' | 'swot' | 'verdict'>('pros_cons');
  const [analysis, setAnalysis] = useState<any>(null);

  // Mock generator to test UI immediately (Hook up to your LLM API here)
  const handleAnalyze = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!decisionInput) return;
    setLoading(true);

    // Simulate API call
    setTimeout(() => {
      setAnalysis({
        summary: `Evaluating trade-offs for: "${decisionInput}"`,
        pros_and_cons: [
          {
            option: "Option A (Take the Leap / Change)",
            pros: [
              { text: "Significantly higher upside potential", weight: 5 },
              { text: "Breaks routine and accelerates learning", weight: 4 },
              { text: "Aligns with long-term ambition", weight: 4 },
            ],
            cons: [
              { text: "Higher short-term financial / emotional risk", weight: -4 },
              { text: "Steep learning curve and transition stress", weight: -3 }
            ]
          },
          {
            option: "Option B (Stay the Course / Safe Path)",
            pros: [
              { text: "Predictable schedule and low baseline stress", weight: 4 },
              { text: "Financial stability and established reputation", weight: 4 }
            ],
            cons: [
              { text: "Potential regret or resentment over time", weight: -5 },
              { text: "Plateau in personal and career growth", weight: -3 }
            ]
          }
        ],
        matrix: [
          { factor: "Financial Upside", optA: "High (Variable)", optB: "Moderate (Fixed)" },
          { factor: "Effort & Stress", optA: "Very High", optB: "Low to Moderate" },
          { factor: "Risk of Regret", optA: "Low", optB: "High" },
          { factor: "Speed of Execution", optA: "2-3 weeks", optB: "Immediate" },
        ],
        swot: {
          strengths: ["Existing baseline skills", "Strong motivation to pivot"],
          weaknesses: ["Limited runway/experience in the new domain"],
          opportunities: ["Entering a high-demand market", "Expanding network"],
          threats: ["Macro-economic uncertainty", "Burnout risk during transition"]
        },
        verdict: {
          recommendation: "Option A is mathematically and emotionally favored if your downside is capped.",
          gutCheck: "If a coin landed on Option B and forced you to stay, would you feel relieved or trapped?",
          tenRule: "In 10 minutes you'll feel anxious. In 10 months you'll be adapted. In 10 years, you won't regret trying."
        }
      });
      setLoading(false);
    }, 1200);
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col items-center p-6 md:p-12 font-sans">
      {/* Header */}
      <header className="max-w-3xl w-full text-center mb-8">
        <div className="inline-flex items-center gap-2 bg-indigo-950/60 border border-indigo-500/30 text-indigo-400 px-4 py-1.5 rounded-full text-sm font-medium mb-4">
          <Scale className="w-4 h-4" /> The Tiebreaker
        </div>
        <h1 className="text-4xl md:text-5xl font-extrabold tracking-tight bg-gradient-to-r from-white via-slate-200 to-slate-400 bg-clip-text text-transparent">
          Clarity when you're stuck in the middle.
        </h1>
        <p className="text-slate-400 mt-2 text-base">
          Enter a decision. Let AI unpack the trade-offs, run the matrix, and break the tie.
        </p>
      </header>

      {/* Input Box */}
      <div className="max-w-3xl w-full bg-slate-900 border border-slate-800 rounded-2xl p-4 shadow-xl mb-8">
        <form onSubmit={handleAnalyze} className="flex flex-col gap-3">
          <textarea
            value={decisionInput}
            onChange={(e) => setDecisionInput(e.target.value)}
            placeholder="e.g. Should I accept an offer at a Series A startup or stay in my stable big tech role?"
            className="w-full bg-slate-950/60 border border-slate-800 rounded-xl p-4 text-slate-100 placeholder-slate-500 focus:outline-none focus:ring-2 focus:ring-indigo-500 resize-none h-24 text-sm md:text-base"
          />
          <div className="flex justify-between items-center pt-1">
            <span className="text-xs text-slate-500">Pro tip: Name both options for better side-by-side results.</span>
            <button
              type="submit"
              disabled={loading || !decisionInput}
              className="inline-flex items-center gap-2 bg-indigo-600 hover:bg-indigo-500 disabled:opacity-50 text-white font-medium px-5 py-2.5 rounded-xl transition-all shadow-md shadow-indigo-600/20 text-sm"
            >
              {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : <Sparkles className="w-4 h-4" />}
              Break the Tie
            </button>
          </div>
        </form>
      </div>

      {/* Analysis Output Section */}
      {analysis && (
        <div className="max-w-3xl w-full bg-slate-900/60 border border-slate-800 rounded-2xl p-6 shadow-2xl backdrop-blur-sm">
          {/* Tabs */}
          <div className="flex border-b border-slate-800 mb-6 gap-2 overflow-x-auto pb-2">
            {[
              { id: 'pros_cons', label: 'Pros & Cons', icon: Scale },
              { id: 'matrix', label: 'Comparison Matrix', icon: Table },
              { id: 'swot', label: 'SWOT Analysis', icon: Grid2X2 },
              { id: 'verdict', label: 'The Tiebreaker Verdict', icon: Sparkles },
            ].map(({ id, label, icon: Icon }) => (
              <button
                key={id}
                onClick={() => setActiveTab(id as any)}
                className={`flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors whitespace-nowrap ${
                  activeTab === id 
                    ? 'bg-indigo-600/20 text-indigo-400 border border-indigo-500/30' 
                    : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
                }`}
              >
                <Icon className="w-4 h-4" />
                {label}
              </button>
            ))}
          </div>

          {/* Tab 1: Pros & Cons */}
          {activeTab === 'pros_cons' && (
            <div className="grid md:grid-cols-2 gap-6">
              {analysis.pros_and_cons.map((item: any, idx: number) => (
                <div key={idx} className="bg-slate-950/40 p-4 rounded-xl border border-slate-800">
                  <h3 className="font-semibold text-slate-200 mb-3 text-sm">{item.option}</h3>
                  <div className="space-y-3">
                    <div className="space-y-1.5">
                      <p className="text-xs font-semibold text-emerald-400 uppercase tracking-wider">Pros</p>
                      {item.pros.map((p: any, i: number) => (
                        <div key={i} className="flex items-start gap-2 text-xs text-slate-300">
                          <CheckCircle2 className="w-3.5 h-3.5 text-emerald-400 mt-0.5 shrink-0" />
                          <span>{p.text} <span className="text-emerald-500/80 font-mono">+{p.weight}</span></span>
                        </div>
                      ))}
                    </div>
                    <div className="space-y-1.5 pt-2 border-t border-slate-800/60">
                      <p className="text-xs font-semibold text-rose-400 uppercase tracking-wider">Cons</p>
                      {item.cons.map((c: any, i: number) => (
                        <div key={i} className="flex items-start gap-2 text-xs text-slate-300">
                          <XCircle className="w-3.5 h-3.5 text-rose-400 mt-0.5 shrink-0" />
                          <span>{c.text} <span className="text-rose-500/80 font-mono">{c.weight}</span></span>
                        </div>
                      ))}
                    </div>
                  </div>
                </div>
              ))}
            </div>
          )}

          {/* Tab 2: Comparison Matrix */}
          {activeTab === 'matrix' && (
            <div className="overflow-x-auto">
              <table className="w-full text-left text-sm">
                <thead>
                  <tr className="border-b border-slate-800 text-slate-400 text-xs uppercase">
                    <th className="py-2.5 px-3">Factor</th>
                    <th className="py-2.5 px-3">Option A</th>
                    <th className="py-2.5 px-3">Option B</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-800/60">
                  {analysis.matrix.map((row: any, i: number) => (
                    <tr key={i} className="hover:bg-slate-800/20">
                      <td className="py-3 px-3 font-medium text-slate-300">{row.factor}</td>
                      <td className="py-3 px-3 text-slate-400">{row.optA}</td>
                      <td className="py-3 px-3 text-slate-400">{row.optB}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}

          {/* Tab 3: SWOT Analysis */}
          {activeTab === 'swot' && (
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <div className="p-4 bg-emerald-950/20 border border-emerald-500/20 rounded-xl">
                <span className="text-xs font-bold uppercase tracking-wider text-emerald-400">Strengths (Internal)</span>
                <ul className="mt-2 space-y-1 text-xs text-slate-300 list-disc list-inside">
                  {analysis.swot.strengths.map((s: string, i: number) => <li key={i}>{s}</li>)}
                </ul>
              </div>
              <div className="p-4 bg-rose-950/20 border border-rose-500/20 rounded-xl">
                <span className="text-xs font-bold uppercase tracking-wider text-rose-400">Weaknesses (Internal)</span>
                <ul className="mt-2 space-y-1 text-xs text-slate-300 list-disc list-inside">
                  {analysis.swot.weaknesses.map((w: string, i: number) => <li key={i}>{w}</li>)}
                </ul>
              </div>
              <div className="p-4 bg-blue-950/20 border border-blue-500/20 rounded-xl">
                <span className="text-xs font-bold uppercase tracking-wider text-blue-400">Opportunities (External)</span>
                <ul className="mt-2 space-y-1 text-xs text-slate-300 list-disc list-inside">
                  {analysis.swot.opportunities.map((o: string, i: number) => <li key={i}>{o}</li>)}
                </ul>
              </div>
              <div className="p-4 bg-amber-950/20 border border-amber-500/20 rounded-xl">
                <span className="text-xs font-bold uppercase tracking-wider text-amber-400">Threats (External)</span>
                <ul className="mt-2 space-y-1 text-xs text-slate-300 list-disc list-inside">
                  {analysis.swot.threats.map((t: string, i: number) => <li key={i}>{t}</li>)}
                </ul>
              </div>
            </div>
          )}

          {/* Tab 4: Verdict */}
          {activeTab === 'verdict' && (
            <div className="space-y-4">
              <div className="p-4 bg-indigo-950/40 border border-indigo-500/30 rounded-xl">
                <h4 className="text-sm font-semibold text-indigo-400 mb-1 flex items-center gap-1.5">
                  <Sparkles className="w-4 h-4" /> The Rational Recommendation
                </h4>
                <p className="text-sm text-slate-200">{analysis.verdict.recommendation}</p>
              </div>

              <div className="grid md:grid-cols-2 gap-4">
                <div className="p-4 bg-slate-950/40 border border-slate-800 rounded-xl">
                  <h5 className="text-xs font-semibold text-slate-400 uppercase tracking-wider mb-2">The Gut Check</h5>
                  <p className="text-xs text-slate-300 italic">"{analysis.verdict.gutCheck}"</p>
                </div>
                <div className="p-4 bg-slate-950/40 border border-slate-800 rounded-xl">
                  <h5 className="text-xs font-semibold text-slate-400 uppercase tracking-wider mb-2">10/10/10 Perspective</h5>
                  <p className="text-xs text-slate-300">{analysis.verdict.tenRule}</p>
                </div>
              </div>
            </div>
          )}
        </div>
      )}
    </div>
  );
}
